cmake_minimum_required(VERSION 3.16)
project(deepspace C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_executable(deepspace src/game.c src/cli.c)
add_executable(test-core src/game.c tests/test_core.c)
foreach(target deepspace test-core)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /fp:strict /UNDEBUG)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off -UNDEBUG)
    target_link_libraries(${target} PRIVATE m)
  endif()
endforeach()
enable_testing()
add_test(NAME core COMMAND test-core)
add_test(NAME seeded-fuzz COMMAND test-core --fuzz)
